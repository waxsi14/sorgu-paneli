  <input type="text" id="tcInput" placeholder="TC girin (örnek: 12345678901)">
  <button onclick="sorgula()">Sorgula</button>

  <div id="sonuc"></div>

  <script>
    // 🔹 Basit örnek veri tabanı
    const veriler = {
      "12345678901": { ad: "Ahmet", soyad: "Yılmaz", yas: 25 },
      "23456789012": { ad: "Mehmet", soyad: "Kaya", yas: 30 },
      "34567890123": { ad: "Ayşe", soyad: "Demir", yas: 28 }
    };

    function sorgula() {
      const tc = document.getElementById("tcInput").value.trim();
      const sonucDiv = document.getElementById("sonuc");

      if (veriler[tc]) {
        const kisi = veriler[tc];
        sonucDiv.innerHTML = `
          ✅ <b>TC:</b> ${tc}<br>
          👤 <b>Ad:</b> ${kisi.ad}<br>
          🧬 <b>Soyad:</b> ${kisi.soyad}<br>
          🎂 <b>Yaş:</b> ${kisi.yas}
        `;
      } else {
        sonucDiv.innerHTML = "❌ Bu TC'ye ait kayıt bulunamadı.";
      }
    }
  </script>
