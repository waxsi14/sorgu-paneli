  <script>
    const qEl = document.getElementById('q');
    const btn = document.getElementById('btn');
    const result = document.getElementById('result');
    const area = document.getElementById('dataArea');
    const save = document.getElementById('save');
    const saveMsg = document.getElementById('saveMsg');

    let DATA = {};

    function loadFromArea(){
      try{
        const parsed = JSON.parse(area.value);
        if (typeof parsed === 'object' && parsed !== null){
          DATA = parsed;
          saveMsg.innerText = 'Veri yüklendi.';
          setTimeout(()=>saveMsg.innerText = '',1200);
        } else {
          saveMsg.innerText = 'JSON bir nesne olmalı.';
          setTimeout(()=>saveMsg.innerText = '',1600);
        }
      }catch(e){
        saveMsg.innerText = 'Geçersiz JSON.';
        setTimeout(()=>saveMsg.innerText = '',1600);
      }
    }

    // ilk yükleme
    loadFromArea();

    function doSearch(){
      const q = qEl.value.trim();
      if (!q) { result.innerText = 'Lütfen bir sorgu gir.'; return; }
      if (Object.prototype.hasOwnProperty.call(DATA,q)){
        result.innerText = 'Sonuç: ' + String(DATA[q]);
      } else {
        result.innerText = 'Sonuç bulunamadı.';
      }
    }

    btn.addEventListener('click', doSearch);
    qEl.addEventListener('keydown', e => { if (e.key === 'Enter') doSearch(); });
    save.addEventListener('click', loadFromArea);

    // İsteğe bağlı: Veriyi localStorage'a kaydet (sayfa yenilense bile kalsın)
    (function(){
      const stored = localStorage.getItem('miniSorguPanelData');
      if (stored){
        area.value = stored;
        loadFromArea();
      }

      save.addEventListener('click', ()=>{
        localStorage.setItem('miniSorguPanelData', area.value);
      });
    })();
  </script>
